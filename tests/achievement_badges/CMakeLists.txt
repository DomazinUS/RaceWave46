cmake_minimum_required(VERSION 3.20)
project(Wr64AchievementBadgeTests LANGUAGES CXX)
get_filename_component(WR64_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(wr64_achievement_badges_test badges.cpp)
target_compile_features(wr64_achievement_badges_test PRIVATE cxx_std_20)
target_include_directories(wr64_achievement_badges_test PRIVATE "${WR64_ROOT}/include")
if(WIN32)
    target_link_libraries(wr64_achievement_badges_test PRIVATE winhttp)
endif()
enable_testing()
add_test(NAME badge-cache-offline COMMAND wr64_achievement_badges_test)
set_tests_properties(badge-cache-offline PROPERTIES TIMEOUT 15)
add_executable(wr64_badge_rounding_test rounding.cpp)
target_compile_features(wr64_badge_rounding_test PRIVATE cxx_std_20)
target_include_directories(wr64_badge_rounding_test PRIVATE "${WR64_ROOT}/include" "${WR64_ROOT}/lib/rt64/src/contrib/stb")
add_test(NAME badge-rounding COMMAND wr64_badge_rounding_test)
set_tests_properties(badge-rounding PROPERTIES TIMEOUT 15)
