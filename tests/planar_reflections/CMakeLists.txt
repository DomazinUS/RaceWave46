cmake_minimum_required(VERSION 3.20)
project(Wr64PlanarReflectionMathTests LANGUAGES CXX)

get_filename_component(WR64_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(wr64_planar_reflection_math math_test.cpp)
target_compile_features(wr64_planar_reflection_math PRIVATE cxx_std_17)
target_include_directories(wr64_planar_reflection_math PRIVATE "${WR64_ROOT}/lib/RT64/src")
if(MSVC)
    target_compile_options(wr64_planar_reflection_math PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(wr64_planar_reflection_math PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME planar_reflection_geometry COMMAND wr64_planar_reflection_math)
set_tests_properties(planar_reflection_geometry PROPERTIES LABELS "cpu;planar-reflections" TIMEOUT 10)
