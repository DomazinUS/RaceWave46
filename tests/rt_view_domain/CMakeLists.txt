cmake_minimum_required(VERSION 3.20)
project(Wr64RTViewDomain LANGUAGES CXX)
enable_testing()
get_filename_component(WR64_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(rt_view_domain_test view_domain_test.cpp)
target_compile_features(rt_view_domain_test PRIVATE cxx_std_17)
target_include_directories(rt_view_domain_test PRIVATE "${WR64_ROOT}/lib/rt64/src")
target_compile_definitions(rt_view_domain_test PRIVATE
    WR64_CAPTURED_MATERIALS="${WR64_ROOT}/tests/sky_split_rebase/captured-cloud-materials.csv")
add_test(NAME rt_native_view_domains COMMAND rt_view_domain_test)
add_executable(rt_existing_sky_test "${WR64_ROOT}/tests/rt_reflections/sky_selection.cpp")
target_compile_features(rt_existing_sky_test PRIVATE cxx_std_17)
add_test(NAME rt_existing_sky_selection COMMAND rt_existing_sky_test)
add_executable(rt_existing_preview_test "${WR64_ROOT}/tests/rt_reflections/preview_selection.cpp")
target_compile_features(rt_existing_preview_test PRIVATE cxx_std_17)
target_include_directories(rt_existing_preview_test PRIVATE "${WR64_ROOT}/lib/rt64/src")
add_test(NAME rt_existing_menu_domains COMMAND rt_existing_preview_test)
