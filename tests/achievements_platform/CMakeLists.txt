cmake_minimum_required(VERSION 3.20)
project(Wr64AchievementsPlatformTests LANGUAGES CXX)
get_filename_component(WR64_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(wr64_achievements_platform_test platform.cpp "${WR64_ROOT}/src/wr64_achievements_platform.cpp")
target_compile_features(wr64_achievements_platform_test PRIVATE cxx_std_20)
target_include_directories(wr64_achievements_platform_test PRIVATE "${WR64_ROOT}/include")
if(WIN32)
    target_link_libraries(wr64_achievements_platform_test PRIVATE winhttp crypt32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WR64_LINUX_PLATFORM REQUIRED IMPORTED_TARGET libcurl libsecret-1)
    target_link_libraries(wr64_achievements_platform_test PRIVATE PkgConfig::WR64_LINUX_PLATFORM)
endif()
