cmake_minimum_required(VERSION 3.20)
project(TitleMenuLayout LANGUAGES C CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
get_filename_component(WR64_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/prepare.py" "${WR64_ROOT}/RecompiledFuncsPoint0/funcs_20.c" "${WR64_ROOT}/config/us.rev1.toml")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare.py" "${WR64_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(title_menu_layout title_menu_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/native_title.c" "${WR64_ROOT}/src/wr64_menu_layout.cpp")
target_compile_features(title_menu_layout PRIVATE cxx_std_20)
target_include_directories(title_menu_layout PRIVATE "${WR64_ROOT}/include" "${WR64_ROOT}/RecompiledFuncsPoint0" "${WR64_ROOT}/lib/N64ModernRuntime/N64Recomp/include")
set(WR64_TITLE_ROM "" CACHE FILEPATH "Private locally stored widescreen USA Rev1 ROM")
if(WR64_TITLE_ROM)
 add_test(NAME title_menu_native_layout COMMAND title_menu_layout "${WR64_TITLE_ROM}")
endif()
